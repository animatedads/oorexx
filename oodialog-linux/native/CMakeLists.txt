cmake_minimum_required(VERSION 3.16)
project(oodialog_qt_listener LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
Python3_add_library(oodialog_qt MODULE WITH_SOABI oodialog_qt.cpp)
target_link_libraries(oodialog_qt PRIVATE Qt6::Widgets)
target_compile_options(oodialog_qt PRIVATE -Wall -Wextra -Wpedantic)
