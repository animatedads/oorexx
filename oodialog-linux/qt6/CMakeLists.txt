cmake_minimum_required(VERSION 3.16)
project(oodialog_qt6 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_executable(oodialog-qt6 main.cpp)
target_link_libraries(oodialog-qt6 PRIVATE Qt6::Widgets)
target_compile_options(oodialog-qt6 PRIVATE -Wall -Wextra -Wpedantic)
