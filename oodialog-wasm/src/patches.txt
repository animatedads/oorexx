--@replace PlainBaseDialog init
-- pbdlg_init of ooDialog.cpp, in Rexx: the object variables it sets belong to this scope.
::method init
  expose dlgData. useStem finished processingLoad
  use arg library = "", resource = 0, , hFile, ownerData
  .oodn~dialogInit(self, library, resource)
  if arg(3, "E") then do
    dlgData. = arg(3)
    useStem = .true
  end
  else useStem = .false
  finished = .false
  processingLoad = .false
  self~childDialogs = .list~new
  self~dataConnection = .array~new(50)
  self~automaticMethods = .queue~new
  self~menuBar = .nil
  self~isLinked = .false
  self~initAutoDetection
  if .constDirUsage == "use only" then self~constDir = .constDir
  else do
    symbols = .directory~new
    self~constDir = symbols
    .oodn~putDefaultSymbols(symbols)
  end
  if arg(4, "E") then self~parseIncludeFile(hFile)
  self~dlgID = .oodn~resolveId(self, resource)
  self~initCode = 0
  if arg(5, "E") then .oodn~dialogOwnerData(self, ownerData)     -- (pbdlg_init: checkDlgType)
  if self~isA(.ResizingAdmin) then .oodn~raSetup(self)     -- (pbdlg_init: allocateResizeInfo, then defineSizing)
  return 0

--@replace PlainBaseDialog finished
-- A plain attribute: the event pump tests it (the DLL kept it as an object variable too).
::attribute finished unguarded

--@replace PlainBaseDialog waitForDialog
-- The DLL's dialog ran on a Windows message loop thread while this method
-- waited (GUARD ON WHEN finished).  Here the waiting thread takes the page's
-- events itself, so event handlers run on it.
::method waitForDialog unguarded
  .oodn~pumpUntilFinished(self)

--@replace UserDialog init
-- userdlg_init of oodUser.cpp, in Rexx (it forwards to the superclass).
::method init
  use arg , includeFile, libOrOwnerData, resourceID, ownerData
  newArgs = .array~new(5)
  if arg(1, "E") then newArgs[3] = arg(1)
  if arg(2, "E") then newArgs[4] = includeFile
  rcBased = self~isA(.RcDialog)
  if \rcBased then rcBased = self~isA(.RcPSPDialog)
  if \rcBased then rcBased = self~isA(.RcControlDialog)
  if rcBased then do
    if arg(3, "E") then newArgs[1] = libOrOwnerData
    if arg(4, "E") then newArgs[2] = resourceID
    else newArgs[2] = ""
    if arg(5, "E") then newArgs[5] = ownerData
  end
  else do
    newArgs[1] = ""
    newArgs[2] = 0
    if arg(3, "E") then newArgs[5] = libOrOwnerData
  end
  forward class (super) arguments (newArgs) continue
  initResult = result
  if initResult == 0 then initResult = self~dynamicInit(0)
  return initResult

  newArgs = .array~new(4)

  if arg(4, "E") then newArgs[2] = includeFile

  forward class (super) arguments (newArgs) continue
  initResult = result
  .oodn~pageInit(self, "RC", pageOpts, connectOpts, expected)
  return initResult

--@replace UserPSPDialog init
-- userpspdlg_init, in Rexx.
::method init
  use arg , includeFile, pageOpts = "", pageTitle = "", fontName = "", fontSize = 0, expected = 200
  newArgs = .array~new(2)
  if arg(1, "E") then newArgs[1] = arg(1)
  if arg(2, "E") then newArgs[2] = includeFile
  forward class (super) arguments (newArgs) continue
  initResult = result
  .oodn~pageInit(self, "USER", pageOpts, "", expected)
  if pageTitle \== "" then self~pageTitle = pageTitle
  if fontName \== "" then self~fontName = fontName
  if fontSize > 0 then self~fontSize = fontSize
  return initResult

--@replace PropertySheetDialog init

  newArgs = .array~new(4)
